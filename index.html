// --- إعدادات الواتساب والموقع --- const siteName = "نظام كاشير الريس (naks8-tech)"; const whatsappNumber = "0791371566"; // 👈 ضع رقمك هنا بالصيغة الدولية (بدون +) function openSubscriptionModal() { const message = encodeURIComponent(`مرحباً، أرغب بالاستفسار عن اشتراكات ${siteName}`); const whatsappUrl = `https://wa.me/${whatsappNumber}?text=${message}`; document.getElementById('whatsappLink').href = whatsappUrl; document.getElementById('subscriptionModal').style.display = 'flex'; } function closeSubscriptionModal() { document.getElementById('subscriptionModal').style.display = 'none'; } // --- إدارة سلة الشراء وتعديل كمية الوجبات --- let cart = []; // إضافة وجبة جديدة للسلة (مثال: وجبة الريس) function addToCart(itemName, price) { let existingItem = cart.find(item => item.name === itemName); if (existingItem) { existingItem.qty += 1; } else { cart.push({ name: itemName, price: price, qty: 1 }); } renderCart(); } // تغيير الكمية يدوياً أو بواسطة الأسهم function updateQuantity(index, newQty) { let parsedQty = parseInt(newQty); if (isNaN(parsedQty) || parsedQty <= 0) { parsedQty = 1; // الحد الأدنى للكمية } cart[index].qty = parsedQty; renderCart(); } // تغيير الكمية بالزيادة/النقصان (+ / -) function changeQtyBy(index, amount) { if (cart[index].qty + amount >= 1) { cart[index].qty += amount; renderCart(); } } // إعادة رسم الفاتورة على الشاشة function renderCart() { const cartContainer = document.getElementById('cartItemsContainer'); // تأكد من مطابقة ID الحاوية في كودك if (!cartContainer) return; cartContainer.innerHTML = ''; let totalAmount = 0; cart.forEach((item, index) => { let itemTotal = item.price * item.qty; totalAmount += itemTotal; cartContainer.innerHTML += `
${item.name}
${item.price} د.أ
${itemTotal.toFixed(3)} د.أ
`; }); // تحديث المجموع النهائي const totalElement = document.getElementById('finalTotal'); if (totalElement) totalElement.innerText = `${totalAmount.toFixed(3)} د.أ`; }
