import sqlite3 import tkinter as tk from tkinter import messagebox, ttk import os import tempfile # ================= 1. قاعدة البيانات ================= def init_db(): conn = sqlite3.connect("pos_system.db") cursor = conn.cursor() # جدول المستخدمين cursor.execute('''CREATE TABLE IF NOT EXISTS users ( id INTEGER PRIMARY KEY AUTOINCREMENT, username TEXT UNIQUE, password TEXT, role TEXT)''') # جدول المنتجات cursor.execute('''CREATE TABLE IF NOT EXISTS products ( id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT, price REAL)''') # إضافة حسابات افتراضية إذا لم تكن موجودة cursor.execute("SELECT * FROM users") if not cursor.fetchall(): cursor.execute("INSERT INTO users (username, password, role) VALUES ('admin', '1234', 'Admin')") cursor.execute("INSERT INTO users (username, password, role) VALUES ('cashier', '1111', 'Cashier')") # إضافة منتجات تجريبية cursor.execute("INSERT INTO products (name, price) VALUES ('وجبة شاورما', 3.50)") cursor.execute("INSERT INTO products (name, price) VALUES ('ساندويش برغر', 2.50)") cursor.execute("INSERT INTO products (name, price) VALUES ('عصير طازج', 1.00)") conn.commit() conn.close() # ================= 2. شاشة تسجيل الدخول ================= class LoginWindow: def __init__(self, root, on_login_success): self.root = root self.root.title("تسجيل الدخول - نظام الكاشير") self.root.geometry("350x250") self.on_login_success = on_login_success tk.Label(root, text="تسجيل الدخول", font=("Arial", 16, "bold")).pack(pady=15) tk.Label(root, text="اسم المستخدم:").pack() self.username_entry = tk.Entry(root) self.username_entry.pack(pady=5) tk.Label(root, text="كلمة السر:").pack() self.password_entry = tk.Entry(root, show="*") self.password_entry.pack(pady=5) tk.Button(root, text="دخول", command=self.login, bg="#4CAF50", fg="white", width=12).pack(pady=15) def login(self): username = self.username_entry.get() password = self.password_entry.get() conn = sqlite3.connect("pos_system.db") cursor = conn.cursor() cursor.execute("SELECT role FROM users WHERE username=? AND password=?", (username, password)) user = cursor.fetchone() conn.close() if user: role = user[0] self.root.destroy() self.on_login_success(username, role) else: messagebox.showerror("خطأ", "اسم المستخدم أو كلمة السر غير صحيحة") # ================= 3. شاشة POS الرئيسية ================= class POSSystem: def __init__(self, root, username, role): self.root = root self.root.title(f"نظام POS - المستخدم: {username} ({role})") self.root.geometry("900x600") self.username = username self.role = role self.cart = [] self.discount = 0.0 # الهيكل العام left_frame = tk.Frame(self.root, width=500, bg="#f0f0f0") left_frame.pack(side=tk.LEFT, fill=tk.BOTH, expand=True, padx=10, pady=10) right_frame = tk.Frame(self.root, width=400) right_frame.pack(side=tk.RIGHT, fill=tk.BOTH, expand=True, padx=10, pady=10) # ----- الجزء الأيمن: قائمة المنتجات ----- tk.Label(right_frame, text="قائمة الوجبات والساندويشات", font=("Arial", 14, "bold")).pack(pady=5) self.product_listbox = tk.Listbox(right_frame, font=("Arial", 12)) self.product_listbox.pack(fill=tk.BOTH, expand=True, pady=5) self.load_products() btn_add = tk.Button(right_frame, text="إضافة للفاتورة", command=self.add_to_cart, bg="#2196F3", fg="white") btn_add.pack(fill=tk.X, pady=5) if self.role == "Admin": btn_edit_price = tk.Button(right_frame, text="تعديل سعر المنتج (مدير)", command=self.edit_product_price, bg="#FF9800", fg="white") btn_edit_price.pack(fill=tk.X, pady=2) # ----- الجزء الأيسر: الفاتورة والحساب ----- tk.Label(left_frame, text="الفاتورة الحالية", font=("Arial", 14, "bold"), bg="#f0f0f0").pack(pady=5) self.cart_tree = ttk.Treeview(left_frame, columns=("Name", "Price"), show="headings") self.cart_tree.heading("Name", text="الصنف") self.cart_tree.heading("Price", text="السعر") self.cart_tree.pack(fill=tk.BOTH, expand=True, pady=5) self.lbl_subtotal = tk.Label(left_frame, text="المجموع الأصلي: 0.00", font=("Arial", 11), bg="#f0f0f0") self.lbl_subtotal.pack(anchor="w") self.lbl_discount = tk.Label(left_frame, text="الخصم: 0.00", font=("Arial", 11), fg="red", bg="#f0f0f0") self.lbl_discount.pack(anchor="w") self.lbl_total = tk.Label(left_frame, text="الصافي النهائي: 0.00", font=("Arial", 14, "bold"), fg="green", bg="#f0f0f0") self.lbl_total.pack(anchor="w", pady=5) btn_discount = tk.Button(left_frame, text="إضافة خصم (يتطلب صلاحية)", command=self.apply_discount_auth, bg="#9C27B0", fg="white") btn_discount.pack(fill=tk.X, pady=5) btn_print = tk.Button(left_frame, text="طباعة الفاتورة وتأديد", command=self.print_receipt, bg="#4CAF50", fg="white", font=("Arial", 12, "bold")) btn_print.pack(fill=tk.X, pady=10) def load_products(self): self.product_listbox.delete(0, tk.END) conn = sqlite3.connect("pos_system.db") cursor = conn.cursor() cursor.execute("SELECT id, name, price FROM products") self.products = cursor.fetchall() conn.close() for p in self.products: self.product_listbox.insert(tk.END, f"{p[1]} - {p[2]:.2f} دينار") def add_to_cart(self): selected = self.product_listbox.curselection() if not selected: messagebox.showwarning("تنبيه", "اختر منتجاً أولاً") return product = self.products[selected[0]] self.cart.append(product) self.cart_tree.insert("", tk.END, values=(product[1], f"{product[2]:.2f}")) self.update_totals() def update_totals(self): subtotal = sum(p[2] for p in self.cart) total = max(0.0, subtotal - self.discount) self.lbl_subtotal.config(text=f"المجموع الأصلي: {subtotal:.2f}") self.lbl_discount.config(text=f"الخصم: {self.discount:.2f}") self.lbl_total.config(text=f"الصافي النهائي: {total:.2f}") def apply_discount_auth(self): # إذا كان المستخدم أدمن يطبق الخصم فوراً، إذا كان كاشير يطلب باس الأدمن if self.role == "Admin": self.ask_discount_amount() else: self.verify_admin_pass(on_success=self.ask_discount_amount) def verify_admin_pass(self, on_success): win = tk.Toplevel(self.root) win.title("تأكيد صلاحية المشرف") win.geometry("300x150") tk.Label(win, text="أدخل كلمة سر المشرف لعمل الخصم:").pack(pady=10) pass_entry = tk.Entry(win, show="*") pass_entry.pack(pady=5) def check(): conn = sqlite3.connect("pos_system.db") cursor = conn.cursor() cursor.execute("SELECT * FROM users WHERE role='Admin' AND password=?", (pass_entry.get(),)) admin = cursor.fetchone() conn.close() if admin: win.destroy() on_success() else: messagebox.showerror("خطأ", "كلمة سر المشرف غير صحيحة!") tk.Button(win, text="موافق", command=check, bg="#4CAF50", fg="white").pack(pady=10) def ask_discount_amount(self): win = tk.Toplevel(self.root) win.title("إدخال الخصم") win.geometry("250x150") tk.Label(win, text="قيمة الخصم:").pack(pady=10) disc_entry = tk.Entry(win) disc_entry.pack(pady=5) def set_disc(): try: self.discount = float(disc_entry.get()) self.update_totals() win.destroy() except ValueError: messagebox.showerror("خطأ", "يرجى إدخال رقم صحيح") tk.Button(win, text="تطبيق", command=set_disc).pack(pady=10) def edit_product_price(self): selected = self.product_listbox.curselection() if not selected: messagebox.showwarning("تنبيه", "اختر منتجاً لتعديل سعره") return product = self.products[selected[0]] win = tk.Toplevel(self.root) win.title("تعديل السعر") win.geometry("250x150") tk.Label(win, text=f"السعر الجديد لـ ({product[1]}):").pack(pady=10) price_entry = tk.Entry(win) price_entry.insert(0, str(product[2])) price_entry.pack(pady=5) def save_price(): try: new_price = float(price_entry.get()) conn = sqlite3.connect("pos_system.db") cursor = conn.cursor() cursor.execute("UPDATE products SET price=? WHERE id=?", (new_price, product[0])) conn.commit() conn.close() win.destroy() self.load_products() except ValueError: messagebox.showerror("خطأ", "السعر غير صحيح") tk.Button(win, text="حفظ", command=save_price).pack(pady=10) def print_receipt(self): if not self.cart: messagebox.showwarning("تنبيه", "الفاتورة فارغة!") return subtotal = sum(p[2] for p in self.cart) total = max(0.0, subtotal - self.discount) # بناء نص الفاتورة للطباعة receipt_text = "====== فاتورة المبيعات ======\n" receipt_text += f"الكاشير: {self.username}\n" receipt_text += "-----------------------------\n" for item in self.cart: receipt_text += f"{item[1]} : {item[2]:.2f}\n" receipt_text += "-----------------------------\n" if self.discount > 0: # شطب السعر الأصلي وإظهار الخصم والسعر الجديد receipt_text += f"السعر الأصلي: ~~ {subtotal:.2f} ~~\n" receipt_text += f"خصم: -{self.discount:.2f}\n" receipt_text += f"السعر الجديد: {total:.2f}\n" else: receipt_text += f"المجموع الكلي: {total:.2f}\n" receipt_text += "=============================\n" receipt_text += " شكراً لزيارتكم! \n" # حفظ الفاتورة في ملف مؤقت وإرسالها للطابعة temp_file = os.path.join(tempfile.gettempdir(), "receipt.txt") with open(temp_file, "w", encoding="utf-8") as f: f.write(receipt_text) # إرسال لأمر الطباعة حسب نظام التشغيل try: if os.name == 'nt': # Windows os.startfile(temp_file, "print") else: # Linux/Mac os.system(f"lpr {temp_file}") messagebox.showinfo("نجاح", "تم إرسال الفاتورة للطابعة بنجاح.") except Exception as e: messagebox.showerror("خطأ طباعة", f"تعذر الطباعة: {e}") # تفريغ السلة بعد الطباعة self.cart.clear() self.discount = 0.0 for item in self.cart_tree.get_children(): self.cart_tree.delete(item) self.update_totals() # ================= 4. التشغيل الرئيسي ================= if __name__ == "__main__": init_db() def start_pos(username, role): pos_root = tk.Tk() POSSystem(pos_root, username, role) pos_root.mainloop() login_root = tk.Tk() LoginWindow(login_root, start_pos) login_root.mainloop()
